async function _cgPerformFetchWithRetries(url, options = {}, maxRetries = 3) {
  let attempt = 0;
  while (attempt <= maxRetries) {
    try {
      const r = await fetch(url, options);
      if (r.status === 429) {
        const backoff = 500 * Math.pow(2, attempt);
        await new Promise(r => setTimeout(r, backoff));
        attempt++;  // Изменяемая переменная через let

      continue;
    }
  }
  return null;
}

  for (let i = 0; i < localStorage.length; i++) {
    const key = localStorage.key(i);

}

## 10. Обработка ошибок (Error Handling)

### Правило: Всегда обрабатывайте ошибки в async/await

**Airbnb Style Guide:**
```javascript
// bad
async function run() {
  await riskyOperation();
}

// good
async function run() {
  try {
    await riskyOperation();
  } catch (error) {
    console.error(error);
  }
}
```

**Код проекта (cache-manager.js):**
```javascript
✅ СООТВЕТСТВУЕТ
get(type, params, ignoreTTL = false) {
  const key = this.generateKey(type, params);
  
  // Memory cache проверка
  if (this.memoryCache.has(key)) {
    const cached = this.memoryCache.get(key);
    if (ignoreTTL || Date.now() < cached.expiry) {
      return cached.data;
    } else {
      this.memoryCache.delete(key);
    }
  }
  
  // localStorage с обработкой ошибок
  try {
    const stored = localStorage.getItem(key);
    if (stored) {
      const parsed = JSON.parse(stored);
      if (ignoreTTL || Date.now() < parsed.expiry) {
        this.memoryCache.set(key, parsed);
        return parsed.data;
      } else {
        localStorage.removeItem(key);
      }
    }
  } catch (e) {
    console.warn('Cache read error:', e);
  }
  
  return null;
}
```
**Вывод:** ✅ try-catch для всех операций с localStorage
